rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAuthenticated() {
      return request.auth != null;
    }

    function isOwner(userId) {
      return isAuthenticated() && request.auth.uid == userId;
    }

    function signedInUid() {
      return request.auth.uid;
    }

    function isParticipant() {
      return isAuthenticated()
        && resource.data.participants is list
        && signedInUid() in resource.data.participants;
    }

    function willBeParticipant() {
      return isAuthenticated()
        && request.resource.data.participants is list
        && signedInUid() in request.resource.data.participants;
    }

    function userRewardFieldsAreSafe() {
      let changed = request.resource.data.diff(resource.data).affectedKeys();
      return (!changed.hasAny(["xp"]) ||
        (
          request.resource.data.xp is number &&
          resource.data.xp is number &&
          request.resource.data.xp >= resource.data.xp &&
          request.resource.data.xp - resource.data.xp <= 500
        )) &&
        (!changed.hasAny(["coins"]) ||
        (
          request.resource.data.coins is number &&
          resource.data.coins is number &&
          request.resource.data.coins >= resource.data.coins &&
          request.resource.data.coins - resource.data.coins <= 50
        ));
    }

    function userProfileUpdateFieldsAreSafe() {
      return request.resource.data.diff(resource.data).affectedKeys().hasOnly([
        "username", "displayName", "photoURL", "division", "zila", "district", "school", "college",
        "className", "groupName", "thana", "examMode", "avatar", "language", "updatedAt",
        "xp", "coins", "level", "rank", "xpToNextLevel", "lastLoginAt", "streak", "maxStreak", "totalStudyTime"
      ]);
    }

    function validParticipantsCreate() {
      return request.resource.data.participants is list
        && request.resource.data.participants.size() == 2
        && request.resource.data.from is string
        && request.resource.data.to is string
        && request.resource.data.from != request.resource.data.to
        && request.resource.data.participants.hasAll([request.resource.data.from, request.resource.data.to])
        && request.resource.data.from in request.resource.data.participants
        && request.resource.data.to in request.resource.data.participants;
    }

    function unchangedFriendCore() {
      return request.resource.data.from == resource.data.from
        && request.resource.data.to == resource.data.to
        && request.resource.data.participants == resource.data.participants;
    }

    function isFriendParticipant() {
      return isAuthenticated()
        && resource.data.participants is list
        && signedInUid() in resource.data.participants;
    }

    match /usernameIndex/{username} {
      allow read: if true;
      allow create: if isAuthenticated()
        && request.resource.data.keys().hasOnly(["uid", "username", "createdAt", "updatedAt"])
        && request.resource.data.uid == signedInUid()
        && request.resource.data.username == username;
      allow update: if isAuthenticated()
        && resource.data.uid == signedInUid()
        && request.resource.data.uid == signedInUid()
        && request.resource.data.username == username
        && request.resource.data.keys().hasOnly(["uid", "username", "createdAt", "updatedAt"]);
      allow delete: if false;
    }

    match /users/{userId} {
      allow read: if isAuthenticated();
      allow create: if isOwner(userId)
        && request.resource.data.uid == userId
        && request.resource.data.xp is number
        && request.resource.data.xp == 0
        && request.resource.data.level is number
        && request.resource.data.level == 1;
      allow update: if isOwner(userId)
        && userProfileUpdateFieldsAreSafe()
        && userRewardFieldsAreSafe();
      allow delete: if isOwner(userId);
    }

    match /userProgress/{progressId} {
      allow read: if isAuthenticated()
        && (progressId.matches(signedInUid() + '_.*') || resource.data.userId == signedInUid());

      allow create: if isAuthenticated()
        && request.resource.data.userId == signedInUid()
        && progressId.matches(signedInUid() + '_.*')
        && request.resource.data.subjectId is string
        && request.resource.data.itemId is string
        && request.resource.data.kind in ["lesson", "quiz"]
        && request.resource.data.rewardClaimed == true;

      allow update: if isAuthenticated()
        && resource.data.userId == signedInUid()
        && request.resource.data.userId == signedInUid()
        && progressId.matches(signedInUid() + '_.*')
        && request.resource.data.subjectId == resource.data.subjectId
        && request.resource.data.itemId == resource.data.itemId
        && request.resource.data.kind == resource.data.kind
        && request.resource.data.rewardClaimed == true;

      allow delete: if false;
    }

    match /writtenProofs/{proofId} {
      allow read: if isAuthenticated() && resource.data.userId == signedInUid();
      allow create: if isAuthenticated()
        && request.resource.data.userId == signedInUid()
        && request.resource.data.subjectId is string
        && request.resource.data.questionId is string
        && request.resource.data.status in ["pending", "approved", "rejected"];
      allow update: if false;
      allow delete: if false;
    }

    match /missionClaims/{claimId} {
      allow read: if isAuthenticated() && resource.data.userId == signedInUid();
      allow create: if isAuthenticated()
        && request.resource.data.keys().hasOnly(["userId", "missionId", "dayKey", "xpReward", "coinReward", "score", "claimedAt"])
        && request.resource.data.userId == signedInUid()
        && request.resource.data.missionId is string
        && request.resource.data.dayKey is string
        && claimId.matches(signedInUid() + '_.*_' + request.resource.data.dayKey)
        && request.resource.data.xpReward is number
        && request.resource.data.coinReward is number
        && request.resource.data.xpReward >= 0
        && request.resource.data.xpReward <= 300
        && request.resource.data.coinReward >= 0
        && request.resource.data.coinReward <= 50;
      allow update, delete: if false;
    }

    match /subjects/{subjectId} {
      allow read: if true;
      allow write: if false;
      match /chapters/{chapterId} {
        allow read: if true;
        allow write: if false;
        match /lessons/{lessonId} {
          allow read: if true;
          allow write: if false;
        }
      }
    }

    match /quizAttempts/{attemptId} {
      allow read: if isAuthenticated() && resource.data.userId == signedInUid();
      allow create: if isAuthenticated()
        && request.resource.data.userId == signedInUid()
        && request.resource.data.subjectId is string
        && request.resource.data.score is number;
      allow update, delete: if false;
    }

    match /leaderboard/{entryId} {
      allow read: if true;
      allow write: if false;
    }

    match /aiHistory/{userId}/{messageId} {
      allow read, write: if isOwner(userId);
    }

    match /missions/{missionId} {
      allow read: if true;
      allow write: if false;
    }

    match /userMissions/{userId}/{missionId} {
      allow read, write: if isOwner(userId);
    }

    match /friendRequests/{requestId} {
      allow read: if isFriendParticipant()
        || (isAuthenticated() && requestId.matches('.*' + signedInUid() + '.*'));

      allow create: if isAuthenticated()
        && request.resource.data.keys().hasOnly(["from", "to", "participants", "status", "blockedBy", "createdAt", "updatedAt"])
        && validParticipantsCreate()
        && request.resource.data.from == signedInUid()
        && request.resource.data.status in ["pending", "blocked"]
        && ((request.resource.data.status == "pending" && request.resource.data.blockedBy == null)
          || (request.resource.data.status == "blocked" && request.resource.data.blockedBy == signedInUid()));

      allow update: if isFriendParticipant()
        && unchangedFriendCore()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(["status", "updatedAt", "cancelledBy", "actionBy", "blockedBy"])
        && (
          (resource.data.status == "pending" && request.resource.data.status == "accepted" && signedInUid() == resource.data.to) ||
          (resource.data.status == "pending" && request.resource.data.status == "cancelled" && signedInUid() == resource.data.from && request.resource.data.cancelledBy == signedInUid()) ||
          (resource.data.status in ["cancelled", "unfriended"] && request.resource.data.status == "pending" && signedInUid() == resource.data.from && request.resource.data.blockedBy == null) ||
          (resource.data.status == "accepted" && request.resource.data.status == "unfriended" && signedInUid() in resource.data.participants && request.resource.data.actionBy == signedInUid()) ||
          (request.resource.data.status == "blocked" && signedInUid() in resource.data.participants && request.resource.data.blockedBy == signedInUid()) ||
          (resource.data.status == "blocked" && resource.data.blockedBy == signedInUid() && request.resource.data.status == "unfriended" && request.resource.data.blockedBy == null && request.resource.data.actionBy == signedInUid())
        );
      allow delete: if false;
    }

    match /challenges/{challengeId} {
      allow read: if isParticipant();
      allow create: if isAuthenticated()
        && request.resource.data.keys().hasOnly(["from", "to", "participants", "status", "createdAt", "updatedAt"])
        && request.resource.data.from == signedInUid()
        && request.resource.data.to is string
        && validParticipantsCreate()
        && willBeParticipant()
        && request.resource.data.status in ["pending", "accepted", "declined"];
      allow update: if isParticipant()
        && request.resource.data.participants == resource.data.participants
        && request.resource.data.from == resource.data.from
        && request.resource.data.to == resource.data.to
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(["status", "updatedAt", "score", "winner"]);
      allow delete: if false;
    }

    match /messages/{messageId} {
      allow read: if isParticipant();
      allow create: if isAuthenticated()
        && request.resource.data.keys().hasOnly(["from", "to", "participants", "content", "createdAt", "read", "readAt"])
        && request.resource.data.from == signedInUid()
        && request.resource.data.to is string
        && request.resource.data.content is string
        && request.resource.data.content.size() > 0
        && request.resource.data.content.size() <= 1000
        && validParticipantsCreate()
        && willBeParticipant();
      allow update: if isParticipant()
        && request.resource.data.participants == resource.data.participants
        && request.resource.data.from == resource.data.from
        && request.resource.data.to == resource.data.to
        && request.resource.data.content == resource.data.content
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(["read", "readAt"])
        && resource.data.to == signedInUid();
      allow delete: if false;
    }

    match /notifications/{userId}/{notifId} {
      allow read, delete: if isOwner(userId);
      allow create: if false;
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
